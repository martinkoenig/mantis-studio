cmake_minimum_required(VERSION 3.24)
project(MantisInstalledSDKConsumer LANGUAGES C)

# A deliberately low client default proves imported usage requirements raise it.
set(CMAKE_C_STANDARD 90)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT DEFINED SDK_VERSION)
  set(SDK_VERSION 0.2.0)
endif()
find_package(MantisPluginSDK ${SDK_VERSION} EXACT CONFIG REQUIRED)
if(NOT MantisPluginSDK_VERSION VERSION_EQUAL SDK_VERSION)
  message(FATAL_ERROR "SDK package version mismatch")
endif()

file(WRITE "${CMAKE_BINARY_DIR}/origins.txt" "config=${MantisPluginSDK_CONFIG}\nversion=${MantisPluginSDK_VERSION}\n")
foreach(target IN ITEMS mantis-sdk-c mantis-sdk-cpp)
  set(imported MantisPluginSDK::${target})
  if(NOT TARGET ${imported})
    message(FATAL_ERROR "Missing imported SDK target: ${imported}")
  endif()
  foreach(property IN ITEMS TYPE IMPORTED INTERFACE_INCLUDE_DIRECTORIES INTERFACE_LINK_LIBRARIES INTERFACE_COMPILE_FEATURES)
    get_target_property(value ${imported} ${property})
    file(APPEND "${CMAKE_BINARY_DIR}/origins.txt" "${imported}.${property}=${value}\n")
    if(property STREQUAL "INTERFACE_INCLUDE_DIRECTORIES" AND DEFINED SDK_FORBIDDEN_ROOT)
      foreach(include IN LISTS value)
        cmake_path(IS_PREFIX SDK_FORBIDDEN_ROOT "${include}" NORMALIZE contaminated)
        if(contaminated)
          message(FATAL_ERROR "SDK source-path export contamination")
        endif()
      endforeach()
    endif()
  endforeach()
endforeach()

get_target_property(c_features MantisPluginSDK::mantis-sdk-c INTERFACE_COMPILE_FEATURES)
get_target_property(cpp_features MantisPluginSDK::mantis-sdk-cpp INTERFACE_COMPILE_FEATURES)
get_target_property(cpp_links MantisPluginSDK::mantis-sdk-cpp INTERFACE_LINK_LIBRARIES)
if(NOT c_features STREQUAL "c_std_11" OR NOT cpp_features STREQUAL "cxx_std_23")
  message(FATAL_ERROR "SDK language usage requirements mismatch")
endif()
if(NOT cpp_links STREQUAL "MantisPluginSDK::mantis-sdk-c")
  message(FATAL_ERROR "SDK C++ dependency mismatch")
endif()

add_executable(existing-c-contract existing-c-abi.c)
target_link_libraries(existing-c-contract PRIVATE MantisPluginSDK::mantis-sdk-c)
add_library(frozen-l2 OBJECT l2/frozen.c)
target_include_directories(frozen-l2 PRIVATE l2/include)
target_compile_features(frozen-l2 PRIVATE c_std_11)
add_executable(installed-l2 l2/compare.c l2/current.c $<TARGET_OBJECTS:frozen-l2>)
target_link_libraries(installed-l2 PRIVATE MantisPluginSDK::mantis-sdk-c)

add_executable(c-consumer c-consumer.c)
target_link_libraries(c-consumer PRIVATE MantisPluginSDK::mantis-sdk-c)
enable_testing()
add_test(NAME installed-c-contract COMMAND c-consumer)
add_test(NAME existing-c-contract COMMAND existing-c-contract)
add_test(NAME frozen-l2-contract COMMAND installed-l2)
set_tests_properties(existing-c-contract frozen-l2-contract PROPERTIES TIMEOUT 10)
if(NOT C_ONLY)
  enable_language(CXX)
  set(CMAKE_CXX_STANDARD 98)
  set(CMAKE_CXX_EXTENSIONS OFF)
  add_executable(cpp-consumer cpp-consumer.cpp)
  target_link_libraries(cpp-consumer PRIVATE MantisPluginSDK::mantis-sdk-cpp)
  add_test(NAME installed-cpp-contract COMMAND cpp-consumer)
endif()
set_tests_properties(installed-c-contract PROPERTIES TIMEOUT 10)
if(NOT C_ONLY)
  set_tests_properties(installed-cpp-contract PROPERTIES TIMEOUT 10)
endif()
