function(mantis_plugin name folder)
 add_library(${name} MODULE ${folder}/${name}.cpp)
 target_link_libraries(${name} PRIVATE mantis-sdk-cpp mantis-options)
 set_target_properties(${name} PROPERTIES PREFIX "")
 file(GENERATE OUTPUT ${CMAKE_BINARY_DIR}/plugins/${name}.json INPUT ${CMAKE_CURRENT_SOURCE_DIR}/${folder}/${name}.json.in)
endfunction()
