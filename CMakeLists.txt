function(mantis_header name)
 add_library(mantis-${name} INTERFACE)
 target_include_directories(mantis-${name} INTERFACE src/${name}/include)
 target_link_libraries(mantis-${name} INTERFACE mantis-options ${ARGN})
endfunction()
function(mantis_library name)
 add_library(mantis-${name} src/${name}/${name}.cpp)
 target_include_directories(mantis-${name} PUBLIC src/${name}/include)
 target_link_libraries(mantis-${name} PUBLIC mantis-options ${ARGN})
endfunction()

function(mantis_plugin name folder)
 add_library(${name} MODULE ${folder}/${name}.cpp)
 target_link_libraries(${name} PRIVATE mantis-sdk-cpp mantis-options)
 set_target_properties(${name} PROPERTIES PREFIX "")
 file(GENERATE OUTPUT ${CMAKE_BINARY_DIR}/plugins/${name}.json INPUT ${CMAKE_CURRENT_SOURCE_DIR}/${folder}/${name}.json.in)
endfunction()
